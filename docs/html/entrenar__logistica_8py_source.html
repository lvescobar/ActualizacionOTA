1"""
2@file entrenar_logistica.py
3@brief Entrena (en el PC) la regresion logistica que usa el ESP32 y escribe sus pesos.
4
5La regresion logistica es un metodo SUPERVISADO: necesita ejemplos con etiqueta
6(0 = normal, 1 = anomalia). El ESP32 no tiene etiquetas, asi que:
7 1. Aqui se generan ejemplos etiquetados en "desviaciones estandar" (z).
8 2. Se entrena el modelo con descenso de gradiente.
9 3. Los pesos se copian a src/detector_logistico.h.
10En el ESP32 solo se calcula la prediccion; la calibracion (media y desviacion)
11se hace con las primeras 30 muestras, como antes.
12
13Uso: python entrenar_logistica.py
14Requiere: pip install numpy
15"""
16import numpy as np
17
18rng = np.random.default_rng(42)
19TOPE = 10.0 # las caracteristicas se limitan a 10 desviaciones (igual que el ESP32)

22def generar(n=4000):
23 """
24 @brief Genera ejemplos etiquetados en espacio z.
25 @param n Cantidad de ejemplos de cada clase.
26 @return (X, y): X con columnas |z_temp| y |z_tacto|; y con 0 normal / 1 anomalia.
27 """
28 # Normal: ruido normal estandar en ambas variables.
29 normal = np.abs(rng.normal(0, 1, (n, 2)))
30 # Anomalia: al menos una variable se aleja mucho (toque del cable, calor, etc.).
31 anom = np.abs(rng.normal(0, 1, (n, 2)))
32 eje = rng.integers(0, 2, n)
33 anom[np.arange(n), eje] = rng.uniform(3.0, 40.0, n)
34 X = np.vstack([normal, anom])
35 y = np.hstack([np.zeros(n), np.ones(n)])
36 return np.minimum(X, TOPE), y

39def sigmoide(z):
40 """@brief Funcion sigmoide: convierte cualquier numero en una probabilidad (0 a 1)."""
41 return 1.0 / (1.0 + np.exp(-z))

44def entrenar(X, y, tasa=0.1, epocas=5000, l2=1e-3):
45 """
46 @brief Descenso de gradiente para regresion logistica con regularizacion L2.
47 @return (w, b): pesos y sesgo.
48 """
49 w = np.zeros(X.shape[1]); b = 0.0
50 for _ in range(epocas):
51 p = sigmoide(X @ w + b)
52 w -= tasa * (X.T @ (p - y) / len(y) + l2 * w)
53 b -= tasa * np.mean(p - y)
54 return w, b
