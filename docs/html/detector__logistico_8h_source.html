64 bool evaluar(float temp, float tacto) {
65 const float x[2] = {temp, tacto};
66
67 // ---- Fase 1: calibracion (algoritmo de Welford: media y varianza) ----
68 if (n_ < MUESTRAS_CALIBRACION) {
69 n_++;
70 for (int i = 0; i < 2; i++) {
71 float d = x[i] - media_[i];
72 media_[i] += d / n_;
73 m2_[i] += d * (x[i] - media_[i]);
74 }
75 if (n_ == MUESTRAS_CALIBRACION) {
76 for (int i = 0; i < 2; i++) {
77 float s = sqrtf(m2_[i] / (MUESTRAS_CALIBRACION - 1));
78 desv_[i] = (s > stdMin_[i]) ? s : stdMin_[i];
79 }
80 }
81 ultimaProb_ = 0.0f;
82 return false;
83 }
84
85 // ---- Fase 2: caracteristicas |z| (limitadas a TOPE_Z) ----
86 float f[2];
87 for (int i = 0; i < 2; i++) {
88 f[i] = fabsf(x[i] - media_[i]) / desv_[i];
89 if (f[i] > TOPE_Z) f[i] = TOPE_Z;
90 }
91
92 // ---- Fase 3: regresion logistica ----
93 float s = SESGO + W_TEMP * f[0] + W_TACTO * f[1];
94 ultimaProb_ = 1.0f / (1.0f + expf(-s));
95 bool anom = ultimaProb_ > UMBRAL_P;
96
97 // ---- Fase 4: solo los datos normales ajustan la media (deriva lenta) ----
98 if (!anom) {
99 for (int i = 0; i < 2; i++) media_[i] += TASA_DERIVA * (x[i] - media_[i]);
100 } else {
101 nAnomalias_++;
102 }
103 return anom;
104 }
