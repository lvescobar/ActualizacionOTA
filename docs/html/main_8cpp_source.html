99void bootCheck() {
100 prefs.begin("fw", false);
101 if (prefs.getBool("pending", false)) {
102 uint8_t t = prefs.getUChar("tries", 0) + 1;
103 prefs.putUChar("tries", t);
104 Serial.printf("Arranque sin validar %u/3\n", t);
105 if (t >= 3) { // 3 arranques sin validar
106 Serial.println("ROLLBACK: se vuelve a la version anterior");
107 prefs.putString("bad", FW_VERSION);
108 prefs.putBool("pending", false);
109 prefs.putUChar("tries", 0);
110 const esp_partition_t* prev = esp_ota_get_next_update_partition(NULL);
111 if (prev && esp_ota_set_boot_partition(prev) == ESP_OK) ESP.restart();

134bool aplicarOTA(const String& url, String hashEsperado) {
135 HTTPClient http;
136 http.begin(url);
137 if (http.GET() != 200) { http.end(); return false; }
138 int total = http.getSize();
139 if (total <= 0 || !Update.begin(total)) { http.end(); return false; }
140
141 mbedtls_sha256_context ctx;
142 mbedtls_sha256_init(&ctx);
143 mbedtls_sha256_starts(&ctx, 0);
144
145 WiFiClient* s = http.getStreamPtr();
146 uint8_t buf[1024];
147 int restante = total;
148 unsigned long ultimo = millis();
149 bool alt = false; unsigned long ta = 0;
150
151 while (restante > 0 && http.connected()) {
152 if (millis() - ta >= 150) { ta = millis(); alt = !alt; setLeds(alt, !alt); } // descargando: se alternan
153 size_t disp = s->available();
154 if (disp) {
155 int n = s->readBytes(buf, min(disp, sizeof(buf)));
156 if (Update.write(buf, n) != (size_t)n) break;
157 mbedtls_sha256_update(&ctx, buf, n);
158 restante -= n;
159 ultimo = millis();
160 } else if (millis() - ultimo > 10000) break; // corte de red
161 delay(1);
162 }
163 http.end();
164
165 if (restante != 0) { Update.abort(); mbedtls_sha256_free(&ctx); Serial.println("OTA incompleta"); return false; }
166
167 uint8_t h[32];
168 mbedtls_sha256_finish(&ctx, h);
169 mbedtls_sha256_free(&ctx);
170 String hex;
171 for (int i = 0; i < 32; i++) { char c[3]; sprintf(c, "%02x", h[i]); hex += c; }
172 hashEsperado.toLowerCase();
173
174 if (hex != hashEsperado) { Update.abort(); Serial.println("HASH INVALIDO, se descarta"); return false; }
175 return Update.end(true); // solo aqui se cambia la particion de arranque
176}

184void checkOTA() {
185 HTTPClient http;
186 http.begin(SERVER + "/version.json");
187 int code = http.GET();
188 Serial.printf("version.json -> HTTP %d\n", code);
189 if (code != 200) { http.end(); return; }
190 JsonDocument doc;
191 if (deserializeJson(doc, http.getString())) { http.end(); return; }
192 http.end();
193
194 String nueva = doc["version"].as<String>();
195 String bin = doc["url"].as<String>();
196 String hash = doc["sha256"].as<String>();
197 if (nueva == FW_VERSION || nueva == prefs.getString("bad", "")) return;
198
199 Serial.println("Actualizando a " + nueva);
200 if (aplicarOTA(bin, hash)) {
201 prefs.putBool("pending", true);
202 prefs.putUChar("tries", 0);
203 delay(200);
204 ESP.restart(); // el OTA reinicia solo, no hace falta pulsar EN
205 }
206}

224void enviar(float t, int tacto, bool anom) {
225 if (WiFi.status() != WL_CONNECTED) return;
226 HTTPClient http;
227 http.begin(SERVER + "/telemetry");
228 http.addHeader("Content-Type", "application/json");
229 String body = String("{\"fw\":\"") + FW_VERSION + "\",\"temp\":" + String(t) +
230 ",\"luz\":" + tacto + ",\"anomalia\":" + (anom ? "true" : "false") +
231 ",\"rssi\":" + WiFi.RSSI() + "}";
232 http.POST(body);
233 http.end();
234}

287#endif

292#endif
293 Serial.printf("T=%.1f tacto=%d anom=%d p=%.2f\n", t, tacto, anom, detector.probabilidad());
294 // Senal debil -> reporta 1 de cada 3 muestras
295 if (WiFi.RSSI() > -80 || ++contEnvio % 3 == 0) enviar(t, tacto, anom);
296 }
297 }

299 // Consulta OTA: cada 30 s (60 s si la senal es debil)
300 unsigned long perOTA = (WiFi.RSSI() < -80) ? 60000 : 30000;
301 if (millis() - tOTA >= perOTA) { tOTA = millis(); conectarWiFi(); checkOTA(); }
